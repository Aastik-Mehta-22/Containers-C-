#include <iostream>
#include <forward_list>

using namespace std;

int main()
{
    
    forward_list<int> flist = {10,20,30,40};
    
   
    flist.push_front(5);
    
    auto it = flist.begin();
    
     flist.insert_after(it ,15); // inserting after operation
    
    flist.erase_after(it); // removing after operation
    
    
    
    flist.sort(); // sort the list
    
    flist.unique(); //  unique elements
    
    flist.reverse(); // reverse the linked list
    
    for(auto ele : flist){
        
        cout<<ele<<" ";
    }
    cout<<endl;
    
    
    
  
    return 0;
}
